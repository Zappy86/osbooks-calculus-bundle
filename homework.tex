\item[\textbf{135.}] $f(x,y)=x+y$, $D=\{(r,\theta)\mid 3\le r\le 5,\; 0\le \theta\le 2\pi\}$.

\item[\textbf{137.}] $f(x,y)=x^4+y^4$, $D=\{(r,\theta)\mid 1\le r\le 2,\; \tfrac{3\pi}{2}\le \theta\le 2\pi\}$.

\item[\textbf{139.}] $f(x,y)=x^4+2x^2y^2+y^4$, where $D=\{(r,\theta)\mid 3\le r\le 4,\; \tfrac{\pi}{3}\le \theta\le \tfrac{2\pi}{3}\}$.

\item[\textbf{141.}] $f(x,y)=\arctan\!\left(\frac{y}{x}\right)$, where $D=\{(r,\theta)\mid 2\le r\le 3,\; \tfrac{\pi}{4}\le \theta\le \tfrac{\pi}{3}\}$.

\item[\textbf{143.}] $\displaystyle \iint_D \big(e^{x^2+y^2}+x^4+2x^2y^2+y^4\big)\arctan\!\left(\frac{y}{x}\right)\,dA$, 
$D=\{(r,\theta)\mid 1\le r\le 2,\; \tfrac{\pi}{4}\le \theta\le \tfrac{\pi}{3}\}$.

\end{enumerate}
