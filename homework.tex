\noindent\textbf{Instructions:} Complete the following exercises from Section~5.3 of
\textit{Calculus Volume 3} by OpenStax (Strang \& Herman). Odd exercises 123--143.\\[0.3em]
\noindent\textit{Source: OpenStax repository source,} \texttt{openstax/osbooks-calculus-bundle}\textit{, Section~5.3.}

\bigskip

\begin{enumerate}[leftmargin=*, itemsep=1.5em]

\item[\textbf{123.}] Find the average value of the function $f(x,y)=3xy$ on the region graphed in the previous exercise.

\item[\textbf{125.}] The region $D$ bounded by $y=\sin x$, $y=1+\sin x$, $x=0$, and $x=\pi/2$ as given in the following figure.

  \webimage{fig125}{https://raw.githubusercontent.com/Zappy86/osbooks-calculus-bundle/copilot/generate-homework-tex-file/media/CNX_Calc_Figure_15_02_202.jpg}

\item[\textbf{127.}] Find the area of the region $D$ given in the previous exercise.

\item[\textbf{129.}] Find the volume of the solid under the graph of the function $f(x,y)=xy+1$ and above the region in the figure in the previous exercise.

\item[\textbf{131.}] Find the signed volume of the solid under the graph of the function $f(x,y)=x+y$ and above the region in the figure from the previous exercise.

\item[\textbf{133.}] The region $D$ bounded by $y=0$ and $y=x^{2}-1$ as given in the following figure.
\begin{center}
  \webimage{fig133}{https://raw.githubusercontent.com/Zappy86/osbooks-calculus-bundle/copilot/generate-homework-tex-file/media/CNX_Calc_Figure_15_02_206.jpg}

\item[\textbf{135.}] Let $D$ be the region bounded above by the curve $y=4-x^{2}$ and below by $y=\cos x$ and the $x$-axis. Explain why $D$ is neither of Type I nor II.

\item[\textbf{137.}] $f(x,y)=1$ and $D=\{(x,y)\mid 0\le x\le \pi/2,\ \sin x\le y\le 1+\sin x\}$.

\item[\textbf{139.}] $f(x,y)=xy$ and $D=\{(x,y)\mid -1\le y\le 1,\ y^{2}-1\le x\le \sqrt{1-y^{2}}\}$.

\item[\textbf{141.}] $\displaystyle \int_{0}^{1}\!\int_{2x}^{3x} \bigl(x+y^{2}\bigr)\,dy\,dx$.

\item[\textbf{143.}] $\displaystyle \int_{e}^{e^{2}}\!\int_{\ln u}^{2} \bigl(v+\ln u\bigr)\,dv\,du$.

\end{enumerate}
