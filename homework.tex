\section*{Section~5.4 --- Triple Integrals}
% ============================================================

\noindent\textbf{Instructions:} Complete the following exercises from Section~5.4 of
\textit{Calculus Volume 3} by OpenStax (Strang \& Herman). Odd exercises 181--195, and
exercise 211.\\[0.3em]
\noindent\textit{Source: OpenStax repository source,} \texttt{Zappy86/osbooks-calculus-bundle}\textit{, Section~5.4.}

\bigskip

\begin{enumerate}[leftmargin=*, itemsep=1.5em]

\item[\textbf{181.}]
Evaluate $\displaystyle\iiint_{B}(2x+3y^{2}+4z^{3})\,dV$, where
\[
  B=\{(x,y,z)\mid 0\le x\le 1,\;0\le y\le 2,\;0\le z\le 3\}.
\]

\item[\textbf{183.}]
Evaluate $\displaystyle\iiint_{B}(x\cos y+z)\,dV$, where
\[
  B=\{(x,y,z)\mid 0\le x\le 1,\;0\le y\le\pi,\;-1\le z\le 1\}.
\]

\item[\textbf{185.}]
Change the order of integration by integrating first with respect to $z$, then $x$,
then $y$:
\[
  \int_{0}^{1}\int_{1}^{2}\int_{2}^{3}(x^{2}+\ln y+z)\,dx\,dy\,dz.
\]

\item[\textbf{187.}]
Change the order of integration by integrating first with respect to $z$, then $x$,
then $y$:
\[
  \int_{-1}^{2}\int_{1}^{3}\int_{0}^{4}\!\left(x^{2}z+\frac{1}{y}\right)dx\,dy\,dz.
\]

\item[\textbf{189.}]
Let $F$, $G$, and $H$ be continuous functions on $[a,b]$, $[c,d]$, and $[e,f]$,
respectively, where $a,b,c,d,e,f$ are real numbers such that $a<b$, $c<d$, and
$e<f$. Show that
\[
  \int_{a}^{b}\int_{c}^{d}\int_{e}^{f}F(x)\,G(y)\,H(z)\,dz\,dy\,dx
  =\left(\int_{a}^{b}F(x)\,dx\right)\!\left(\int_{c}^{d}G(y)\,dy\right)\!\left(\int_{e}^{f}H(z)\,dz\right).
\]

\item[\textbf{191.}]
Evaluate $\displaystyle\iiint_{E}(2x+5y+7z)\,dV$, where
\[
  E=\{(x,y,z)\mid 0\le x\le 1,\;0\le y\le -x+1,\;1\le z\le 2\}.
\]

\item[\textbf{193.}]
Evaluate $\displaystyle\iiint_{E}(\sin x+\sin y)\,dV$, where
\[
  E=\!\left\{(x,y,z)\;\middle|\;0\le x\le\frac{\pi}{2},\;
    {-\cos x}\le y\le\cos x,\;-1\le z\le 1\right\}.
\]

\item[\textbf{195.}]
Evaluate $\displaystyle\iiint_{E}(x+2yz)\,dV$, where
\[
  E=\{(x,y,z)\mid 0\le x\le 1,\;0\le y\le x,\;0\le z\le 5-x-y\}.
\]

\item[\textbf{211.}]
The solid $E$ bounded by $y^{2}+z^{2}=9$, $z=0$, $x=0$, and $x=5$ is shown in the
following figure. Evaluate the integral $\displaystyle\iiint_{E}z\,dV$ by integrating
first with respect to $z$, then $y$, and then $x$.
\begin{center}
  \webimage{fig211}{https://raw.githubusercontent.com/Zappy86/osbooks-calculus-bundle/main/media/CNX_Calc_Figure_15_04_201.jpg}
\end{center}

\end{enumerate}

\section*{Section~5.5 --- Triple Integrals in Cylindrical and Spherical Coordinates}
% ============================================================

\noindent\textbf{Instructions:} Complete the following exercises from Section~5.5 of
\textit{Calculus Volume 3} by OpenStax (Strang \& Herman). Exercises 241, 245, 253,
255, 269, 277.\\[0.3em]
\noindent\textit{Source: OpenStax repository source,} \texttt{Zappy86/osbooks-calculus-bundle}\textit{, Section~5.5.}

\bigskip

\begin{enumerate}[leftmargin=*, itemsep=1.5em]

\item[\textbf{241.}]
Evaluate the triple integral $\displaystyle\iiint_{B}f(x,y,z)\,dV$ over the solid $B$.

\smallskip
$f(x,y,z)=z$,
\[
  B=\{(x,y,z)\mid x^{2}+y^{2}\le 9,\;x\ge 0,\;y\ge 0,\;0\le z\le 1\}.
\]
\begin{center}
  \webimage{fig241}{https://raw.githubusercontent.com/Zappy86/osbooks-calculus-bundle/main/media/CNX_Calc_Figure_15_05_201.jpg}
\end{center}

\item[\textbf{245.}]
Evaluate the triple integral $\displaystyle\iiint_{B}f(x,y,z)\,dV$ over the solid $B$.

\smallskip
$f(x,y,z)=e^{\sqrt{x^{2}+y^{2}}}$,
\[
  B=\{(x,y,z)\mid 1\le x^{2}+y^{2}\le 4,\;y\le 0,\;x\le y\sqrt{3},\;2\le z\le 3\}.
\]

\item[\textbf{253.}]
The function $g$ and region $E$ are given in rectangular coordinates.
\begin{enumerate}[label=(\alph*)]
  \item Express the region $E$ and the function $g$ in cylindrical coordinates.
        Let $f(r,\theta,z)$ be the corresponding function in cylindrical coordinates.
  \item Convert the integral $\displaystyle\iiint_{E}g(x,y,z)\,dV$ to cylindrical
        coordinates and evaluate it.
\end{enumerate}

\smallskip
$g(x,y,z)=\dfrac{1}{x+3}$,
\[
  E=\{(x,y,z)\mid 0\le x^{2}+y^{2}\le 9,\;x\ge 0,\;y\ge 0,\;0\le z\le x+3\}.
\]

\item[\textbf{255.}]
The function $g$ and region $E$ are given in rectangular coordinates.
\begin{enumerate}[label=(\alph*)]
  \item Express the region $E$ and the function $g$ in cylindrical coordinates.
        Let $f(r,\theta,z)$ be the corresponding function in cylindrical coordinates.
  \item Convert the integral $\displaystyle\iiint_{E}g(x,y,z)\,dV$ to cylindrical
        coordinates and evaluate it.
\end{enumerate}

\smallskip
$g(x,y,z)=x$,
\[
  E=\{(x,y,z)\mid 1\le y^{2}+z^{2}\le 9,\;0\le x\le 9-y^{2}-z^{2}\}.
\]

\item[\textbf{269.}]
Evaluate the triple integral $\displaystyle\iiint_{B}f(x,y,z)\,dV$ over the solid $B$.

\smallskip
$f(x,y,z)=1$,
\[
  B=\{(x,y,z)\mid x^{2}+y^{2}+z^{2}\le 90,\;z\ge 0\}.
\]
\begin{center}
  \webimage{fig269}{https://raw.githubusercontent.com/Zappy86/osbooks-calculus-bundle/main/media/CNX_Calc_Figure_15_05_205.jpg}
\end{center}

\item[\textbf{277.}]
The function $g$ and region $E$ are given in rectangular coordinates.
\begin{enumerate}[label=(\alph*)]
  \item Express the region $E$ and the function $g$ in spherical coordinates.
        Let $f(\rho,\theta,\varphi)$ be the corresponding function in spherical
        coordinates.
  \item Convert the integral $\displaystyle\iiint_{E}g(x,y,z)\,dV$ to spherical
        coordinates and evaluate it.
\end{enumerate}

\smallskip
$g(x,y,z)=x+y$;
\[
  E=\{(x,y,z)\mid 1\le x^{2}+y^{2}+z^{2}\le 4,\;z\ge 0,\;y\ge 0\}.
\]

\end{enumerate}
