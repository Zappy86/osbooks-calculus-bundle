\noindent\textbf{Instructions:} Complete the following exercises from Section~6.1 of
\textit{Calculus Volume 3} by OpenStax (Strang \& Herman). Exercises 2--5, 9, 17, 23, 32--34.\\[0.3em]
\noindent\textit{Source: OpenStax repository source,} \texttt{Zappy86/osbooks-calculus-bundle}\textit{, Section~6.1.}

\bigskip

\begin{enumerate}[leftmargin=*, itemsep=1.5em]

\item[]\textit{For exercises 2--4, determine whether the statement is \textbf{true or false}.}

\item[\textbf{2.}]
Vector field $\mathbf{F} = \langle 3x^2,\,1\rangle$ is a gradient field for both
$\phi_1(x,y) = x^3 + y$ and $\phi_2(x,y) = y + x^3 + 100$.

\item[\textbf{3.}]
Vector field $\mathbf{F} = \dfrac{\langle y,\,x\rangle}{\sqrt{x^2+y^2}}$ is constant
in direction and magnitude on a unit circle.

\item[\textbf{4.}]
Vector field $\mathbf{F} = \dfrac{\langle y,\,x\rangle}{\sqrt{x^2+y^2}}$ is neither a
radial field nor a rotation.

\section*{Section~6.2 --- Line Integrals}
% ============================================================

\noindent\textbf{Instructions:} Complete the following exercises from Section~6.2 of
\textit{Calculus Volume 3} by OpenStax (Strang \& Herman). Exercises 39--43, 51, 54,
57, 62, 67.\\[0.3em]
\noindent\textit{Source: OpenStax repository source,} \texttt{Zappy86/osbooks-calculus-bundle}\textit{, Section~6.2.}

\bigskip

\begin{enumerate}[leftmargin=*, itemsep=1.5em]

\item[]\textit{For exercises 39--43, determine whether the statement is \textbf{true or false}.}

\item[\textbf{39.}]
\textit{True or False?}\enspace Line integral $\displaystyle\int_C f(x,y)\,ds$ is equal to a
definite integral if $C$ is a smooth curve defined on $[a,b]$ and if function $f$ is
continuous on some region that contains curve $C$.

\item[\textbf{40.}]
\textit{True or False?}\enspace Vector functions $\mathbf{r}_1 = t\,\mathbf{i} + t^2\,\mathbf{j}$,
$0\le t\le 1$, and $\mathbf{r}_2 = (1-t)\,\mathbf{i} + (1-t)^2\,\mathbf{j}$, $0\le t\le 1$,
define the same oriented curve.

\item[\textbf{41.}]
\textit{True or False?}\enspace
$\displaystyle\int_{-C}(P\,dx+Q\,dy) = \int_C(P\,dx-Q\,dy)$

\item[\textbf{42.}]
\textit{True or False?}\enspace A piecewise smooth curve $C$ consists of a finite number of
smooth curves that are joined together end to end.

\item[\textbf{43.}]
\textit{True or False?}\enspace If $C$ is given by $x(t)=t$, $y(t)=t$, $0\le t\le 1$, then
\[
  \int_C xy\,ds = \int_0^1 t^2\,dt.
\]

\item[]\textit{For exercises 51 and 54, find the work done.}

\item[\textbf{51.}]
Find the work done by force field
$\mathbf{F}(x,y,z) = -\dfrac{1}{2}x\,\mathbf{i} - \dfrac{1}{2}y\,\mathbf{j} + \dfrac{1}{4}\,\mathbf{k}$
on a particle as it moves along the helix
$\mathbf{r}(t) = \cos t\,\mathbf{i} + \sin t\,\mathbf{j} + t\,\mathbf{k}$
from point $(1,0,0)$ to point $(-1,0,3\pi)$.

\item[\textbf{54.}]
Find the mass of a wire in the shape of a circle of radius~$2$ centered at $(3,4)$
with linear mass density $\rho(x,y)=y^2$.

\item[]\textit{For exercises 57 and 62, evaluate the line integral.}

\item[\textbf{57.}]
Evaluate $\displaystyle\int_C yz\,dx + xz\,dy + xy\,dz$ over the line segment from
$(1,1,1)$ to $(3,2,0)$.

\item[\textbf{62.}]
Evaluate line integral $\displaystyle\int_C (2x-y)\,dx + (x+3y)\,dy$, where $C$ lies
along the $x$-axis from $x=0$ to $x=5$.

\item[]\textit{For exercise 67, find the work done by force field $\mathbf{F}$ on an object
moving along the indicated path.}

\item[\textbf{67.}]
$\mathbf{F}(x,y,z) = x\,\mathbf{i} + y\,\mathbf{j} - 5z\,\mathbf{k}$

\smallskip
$C\colon\;\mathbf{r}(t) = 2\cos t\,\mathbf{i} + 2\sin t\,\mathbf{j} + t\,\mathbf{k},
\quad 0\le t\le 2\pi$

\end{enumerate}
